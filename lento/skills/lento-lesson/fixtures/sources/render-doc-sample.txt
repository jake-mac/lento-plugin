// A second function, deliberately far from line 3: the ±3-line window must not
// reach it, so a quote taken from here cannot confirm a claim citing line 3.
export function ditherFloydSteinberg(pixels, width) {
  const out = new Uint8Array(pixels.length);
  for (let i = 0; i < pixels.length; i++) {
    out[i] = pixels[i] > 128 ? 1 : 0; // no error diffusion in this sample
  }
  return out;
}
